# hal-voice — Python dependencies
#
# REQUIRES PYTHON 3.10–3.12. Not 3.13+: numpy 1.x publishes no wheels past cp312, and
# several of the audio libraries below are still on numpy 1.x. setup-voice.sh finds a
# suitable interpreter for you and refuses to build against one that cannot work.
#
# Wake-word support is NOT here — see requirements-wake.txt. It is optional, off by
# default, and its tflite-runtime dependency is the least portable thing in the project.

# --- Core / config ---
PyYAML>=6.0
numpy>=1.24,<2.0          # GPT-SoVITS + many audio libs are happier on numpy 1.x

# --- Audio capture / playback ---
sounddevice>=0.4.6        # PortAudio bindings for mic capture + playback
soundfile>=0.12.1         # read/write WAV
webrtcvad-wheels>=2.0.14  # voice-activity detection. The '-wheels' fork ships prebuilt
                          # binaries; plain `webrtcvad` needs a C compiler and often fails.

# --- Speech to text (CPU) ---
faster-whisper>=1.0.0     # CTranslate2 Whisper — fast int8 on CPU

# --- Push-to-talk key capture ---
pynput>=1.7.6             # cross-session key listener for push-to-talk

# --- HAL voice FX (DSP) ---
scipy>=1.10               # filtering / resampling
pyloudnorm>=0.1.1         # LUFS loudness normalization
pedalboard>=0.9.0         # Spotify's DSP: reverb, EQ, distortion, pitch (the "HAL effect")

# --- MCP server (speak/listen tools) ---
mcp>=1.2.0                # Model Context Protocol SDK (FastMCP)

# --- Red-eye UI ---
pygame>=2.5.0

# --- HTTP to the GPT-SoVITS api_v2 server ---
requests>=2.31.0

# NOTE: GPT-SoVITS itself is NOT a pip dependency here — it is installed as a
# separate inference server by setup-voice.sh (it has its own heavy torch stack).
# hal-voice talks to it over HTTP, so the two environments stay cleanly separated.
