# hal-voice — OPTIONAL wake-word dependencies ("Hey HAL", hands-free activation).
#
# Kept out of requirements.txt on purpose. Wake word is disabled by default
# (activation.mode: push_to_talk) and detection is unreliable — push-to-talk is the
# supported way in. More practically, tflite-runtime is the least portable dependency
# in the project: it publishes no wheels for Python 3.13+ and none at all for some
# platforms, so having it in the core list breaks installs for people who will never
# turn the feature on.
#
# Install only if you want to experiment:
#     ./setup-voice.sh --with-wake
#     # or, into an existing venv:
#     .venv/bin/pip install -r requirements-wake.txt
#
# Then set activation.mode to wake_word (or both) in config.yaml.

openwakeword>=0.6.0       # ONNX/tflite wake-word models, CPU
